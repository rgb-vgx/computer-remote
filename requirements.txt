# Dependencies cho remote desktop MVP.
# Lưu ý: HOST (Kubuntu) không cần PySide6; CLIENT (Windows) không cần mss/pynput.
# Để chung một file cho đơn giản giai đoạn học — ai muốn gọn có thể tỉa bớt.

mss                    # chụp màn hình (host)
opencv-python-headless # encode/decode JPEG + resize (cả host & client; không cần highgui/Qt)
numpy                  # buffer ảnh (cả host & client)
PySide6                # GUI viewer (client)
pynput                 # inject chuột vào X11 (host)
python-xlib            # inject bàn phím qua XSendEvent (host; pynput cũng cần)
